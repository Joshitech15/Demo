Usage Example

Here's how easy it is to implement the DataTable in your project:

// Basic initialization
const dataTable = new DataTable({
    container: '#datatable-container',
    apiUrl: 'api.php',
    recordsPerPage: 10,
    columns: [
        { key: 'id', title: 'ID' },
        { key: 'first_name', title: 'First Name' },
        { key: 'last_name', title: 'Last Name' },
        { key: 'email', title: 'Email' },
        { key: 'company', title: 'Company' },
        { 
            key: 'salary', 
            title: 'Salary',
            formatter: (value) => new Intl.NumberFormat('en-US', {
                style: 'currency',
                currency: 'USD'
            }).format(value)
        }
    ]
});

// Advanced configuration
const advancedTable = new DataTable({
    container: '#advanced-table',
    apiUrl: 'api.php',
    recordsPerPage: 25,
    searchDelay: 500,
    showSearch: true,
    showRecordsPerPage: true,
    showPaginationInfo: true,
    sortable: true,
    responsive: true,
    loadingText: 'Loading data...',
    noDataText: 'No records found',
    columns: [
        // ... column definitions
    ]
});

API Documentation

Configuration Options

{
    container: '#datatable-container',    // CSS selector for container
    apiUrl: 'api.php',                   // API endpoint URL
    recordsPerPage: 10,                  // Default records per page
    searchDelay: 300,                    // Search input delay (ms)
    showSearch: true,                    // Show search input
    showRecordsPerPage: true,            // Show records per page selector
    showPaginationInfo: true,            // Show pagination information
    columns: [],                         // Column definitions
    sortable: true,                      // Enable column sorting
    responsive: true,                    // Enable responsive design
    loadingText: 'Loading...',           // Loading state text
    noDataText: 'No data available'      // No data state text
}

API Response Format

{
    "success": true,
    "data": [
        {
            "id": 1,
            "first_name": "John",
            "last_name": "Doe",
            "email": "john.doe@example.com",
            "company": "Tech Corp",
            "salary": 75000
        }
        // ... more records
    ],
    "pagination": {
        "current_page": 1,
        "total_pages": 15,
        "total_records": 150,
        "records_per_page": 10,
        "start_record": 1,
        "end_record": 10,
        "has_previous": false,
        "has_next": true,
        "search_term": ""
    },
    "page_numbers": [1, 2, 3, 4, 5]
}

Methods

// Navigate to a specific page
dataTable.goToPage(3);

// Search data
dataTable.search('john');

// Sort by column
dataTable.sort('email');

// Refresh data
dataTable.refresh();

// Update configuration
dataTable.updateConfig({
    recordsPerPage: 25,
    showSearch: false
});

// Destroy the table
dataTable.destroy();
